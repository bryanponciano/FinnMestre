$isAdmin = isset($_SESSION['usuario_email']) && $_SESSION['usuario_email'] === 'poncianobryan988@gmail.com';

// Configurações de contato
try {
    $stmt = $pdo->query("SELECT valor FROM configuracoes WHERE chave = 'whatsapp_suporte'");
    $result = $stmt->fetch();
    $whatsappNumero = $result ? $result['valor'] : '';
} catch (Exception $e) {
    $whatsappNumero = '';
}

// Salvar configuração de WhatsApp (apenas admin)
if ($_SERVER['REQUEST_METHOD'] === 'POST' && $isAdmin && isset($_POST['whatsapp_numero'])) {
    $numero = preg_replace('/\D/', '', $_POST['whatsapp_numero']);
    try {
        $stmt = $pdo->prepare("INSERT INTO configuracoes (chave, valor, tipo, descricao) 
            VALUES ('whatsapp_suporte', ?, 'string', 'Número do WhatsApp de suporte') 
            ON DUPLICATE KEY UPDATE valor = ?");
        $stmt->execute([$numero, $numero]);
        $whatsappNumero = $numero;
        $mensagemSucesso = 'Número de WhatsApp atualizado!';
    } catch (Exception $e) {
        $mensagemErro = 'Erro ao salvar número.';
    }
}

$emailSuporte = 'suporte@finnmestre.com.br';

// Processar envio de email pelo modal
$mensagemEmail = '';
$tipoMensagemEmail = '';
if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['enviar_email'])) {
    $nomeContato = trim($_POST['nome_contato'] ?? '');
    $emailContato = trim($_POST['email_contato'] ?? '');
    $msgContato = trim($_POST['mensagem_contato'] ?? '');

    if (!empty($nomeContato) && !empty($emailContato) && !empty($msgContato)) {
        $assunto = "Novo Contato de Suporte - FinMestre";
        $corpo = "Nome: $nomeContato\nE-mail: $emailContato\n\nMensagem:\n$msgContato";
        // O cabeçalho From DEVE ser um email do próprio domínio na Hostinger para evitar bloqueios de SPAM
        $headers = "From: $emailSuporte\r\n"; 
        $headers .= "Reply-To: $emailContato\r\n";
        $headers .= "X-Mailer: PHP/" . phpversion();

        if (@mail($emailSuporte, $assunto, $corpo, $headers)) {
            $mensagemEmail = 'Sua mensagem foi enviada com sucesso! Entraremos em contato em breve.';
            $tipoMensagemEmail = 'success';
        } else {
            $mensagemEmail = 'Erro ao enviar a mensagem. Por favor, tente novamente mais tarde.';
            $tipoMensagemEmail = 'danger';
        }
    } else {
        $mensagemEmail = 'Por favor, preencha todos os campos.';
        $tipoMensagemEmail = 'danger';
    }
}
